set(mupdf_backend_sources generator_mupdf.cpp annots.cpp latexappearance.cpp latexappearancepolicy.cpp pageediting.cpp pageimport.cpp metadata.cpp navigation.cpp linkediting.cpp ocr.cpp export.cpp forms.cpp resources.cpp)
if(BUILD_DESKTOP)
    install(PROGRAMS ../poppler/okularApplication_pdf.desktop DESTINATION ${KDE_INSTALL_APPDIR})
endif()
if(BUILD_MOBILE)
    install(PROGRAMS ../poppler/org.kde.mobile.okular_pdf.desktop DESTINATION ${KDE_INSTALL_APPDIR})
endif()
okular_add_generator(okularGenerator_mupdf ${mupdf_backend_sources})
target_link_libraries(okularGenerator_mupdf PRIVATE okularcore MuPDF::MuPDF Qt6::Gui Qt6::Xml Qt6::PrintSupport KF6::I18n)
add_library(pdf_page_bounds STATIC pdfpagebounds.cpp)
set_target_properties(pdf_page_bounds PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(pdf_page_bounds PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../poppler")
target_link_libraries(pdf_page_bounds PRIVATE MuPDF::MuPDF Qt6::Core)
add_executable(mupdf_backend_test EXCLUDE_FROM_ALL autotests/backendtest.cpp ${mupdf_backend_sources})
if(BUILD_TESTING)
    set_target_properties(mupdf_backend_test PROPERTIES EXCLUDE_FROM_ALL FALSE)
    add_test(NAME mupdf_backend_test COMMAND mupdf_backend_test)
    set_tests_properties(mupdf_backend_test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
target_link_libraries(mupdf_backend_test PRIVATE okularcore pdf_page_bounds MuPDF::MuPDF Qt6::Test Qt6::Gui Qt6::Xml Qt6::PrintSupport KF6::I18n)
if(HAVE_TESSERACT)
    foreach(target IN ITEMS okularGenerator_mupdf mupdf_backend_test)
        target_compile_definitions(${target} PRIVATE HAVE_TESSERACT=1)
        if(WIN32)
            # Upstream's MSVC archive already embeds Tesseract. Use its matching
            # public headers and avoid importing the SDK's second implementation.
            target_include_directories(${target} PRIVATE "${MuPDF_INCLUDE_DIR}/mupdf-thirdparty")
        else()
            target_link_libraries(${target} PRIVATE Tesseract::Tesseract)
        endif()
    endforeach()
endif()
